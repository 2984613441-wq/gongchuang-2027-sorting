#### 4.2 Python测试脚本

创建测试脚本 `test_turntable.py`：

```python
#!/usr/bin/env python3
"""转盘舵机测试脚本"""

import serial
import time
import sys

PORT = '/dev/ttyAMA3'
BAUD = 1000000
SERVO_ID = 1

# feetech协议常量（需要根据实际协议调整）
HEADER = [0xFF, 0xFF]
CMD_WRITE_POS = 0x03
CMD_READ_POS = 0x02

def calculate_checksum(data):
    """计算校验和"""
    return (~sum(data[2:])) & 0xFF

def move_servo(ser, servo_id, position, speed=100):
    """发送位置命令
    
    Args:
        servo_id: 舵机ID
        position: 目标位置（0~4095）
        speed: 速度（0~4095）
    """
    # 构造数据包（示例，需要根据实际协议调整）
    length = 7
    cmd = CMD_WRITE_POS
    pos_low = position & 0xFF
    pos_high = (position >> 8) & 0xFF
    speed_low = speed & 0xFF
    speed_high = (speed >> 8) & 0xFF
    
    packet = HEADER + [servo_id, length, cmd, pos_low, pos_high, speed_low, speed_high]
    checksum = calculate_checksum(packet)
    packet.append(checksum)
    
    ser.write(bytes(packet))
    ser.flush()

def test_turntable():
    """测试转盘A/B侧切换"""
    try:
        ser = serial.Serial(PORT, BAUD, timeout=1)
        print(f"已连接到 {PORT} @ {BAUD}")
        time.sleep(0.5)
        
        # 位置计算（根据sorting_config.ini）
        # center_us = 1500, us_per_90deg = 655
        # A侧 = 1500 - 655 = 845µs
        # B侧 = 1500 + 655 = 2155µs
        # 需要转换为舵机位置值（0~4095）
        
        # 假设舵机角度范围0~270°对应0~4095
        # 845µs  对应 约 45°  对应 位置 682
        # 2155µs 对应 约 225° 对应 位置 3413
        POS_A = 682   # A侧位置（需实测调整）
        POS_B = 3413  # B侧位置（需实测调整）
        
        cycle = 0
        while True:
            cycle += 1
            
            # 转到A侧
            print(f"\n[周期 {cycle}] 转到A侧 (位置 {POS_A})")
            move_servo(ser, SERVO_ID, POS_A, speed=500)
            time.sleep(2.0)
            
            # 转到B侧
            print(f"[周期 {cycle}] 转到B侧 (位置 {POS_B})")
            move_servo(ser, SERVO_ID, POS_B, speed=500)
            time.sleep(2.0)
    
    except serial.SerialException as e:
        print(f"串口错误: {e}")
        sys.exit(1)
    
    except KeyboardInterrupt:
        print("\n\n测试停止")
        ser.close()
        sys.exit(0)
    
    finally:
        if 'ser' in locals() and ser.is_open:
            ser.close()

if __name__ == '__main__':
    print("=" * 50)
    print("转盘舵机测试 - A/B侧切换")
    print("=" * 50)
    print(f"串口: {PORT}")
    print(f"波特率: {BAUD}")
    print(f"舵机ID: {SERVO_ID}")
    print("按 Ctrl+C 停止测试")
    print("=" * 50)
    
    test_turntable()
```

#### 5.2 手动测试（简化版）

如果生产链太复杂，可以直接通过串口发送测试命令：

```python
#!/usr/bin/env python3
"""XY电机简单测试"""

import serial
import time

PORT = '/dev/serial0'
BAUD = 115200

ser = serial.Serial(PORT, BAUD, timeout=1)

# 发送ASCII测试命令（如果固件支持）
commands = [
    b"MOVE_X 100\n",   # X轴移动100步
    b"MOVE_Y 100\n",   # Y轴移动100步
    b"MOVE_X -100\n",  # X轴返回
    b"MOVE_Y -100\n",  # Y轴返回
]

for cmd in commands:
    print(f"发送: {cmd.decode().strip()}")
    ser.write(cmd)
    ser.flush()
    time.sleep(2)
    
    if ser.in_waiting > 0:
        response = ser.read(ser.in_waiting).decode('utf-8', errors='ignore')
        print(f"收到: {response.strip()}")

ser.close()
```

**⚠️ 注意**：
- 固件可能不支持ASCII命令
- 如果不响应，需要发送二进制ITEM_TASK帧
- 这需要实现完整的协议编码
