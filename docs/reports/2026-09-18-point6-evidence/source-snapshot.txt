 186:     def _accept_clients(self, logger: logging.Logger) -> None:
 187:         """接受客户端连接（独立线程）"""
 188:         while self.running and self.server:
 189:             try:
 190:                 client, addr = self.server.accept()
 191:                 logger.info(f"[MONITOR] 客户端连接: {addr}")
 192:                 with self.lock:
 193:                     self.clients.append(client)
 194:             except socket.timeout:
 195:                 continue
 196:             except Exception as e:
 197:                 if self.running:
 198:                     logger.debug(f"[MONITOR] 接受连接失败: {e}")

 202:         """向所有客户端广播数据"""

 208:         except Exception:
 209:             return

 213:             for client in self.clients:

 291: def load_fixed_pick_point(path: str | Path) -> tuple[int, int] | None:
 292:     """从 sorting_config.ini 的 [PickPoint] 读**固定抓取点**（单位 mm → 0.1mm）。
 293: 
 294:     为什么要有这个（2026-09-18 用户裁定）：
 295:     用户决定**每次把物块放在同一个位置**。那样抓取点在装置坐标里就是一个
 296:     **常量**，不再需要像素→毫米标定 —— 视觉只负责"这是什么件"（分类/计数），
 297:     不负责"它在哪"。
 298: 
 299:     这**不是**"猜一个坐标"（项目明令禁止的那种）：
 300:     · 猜坐标 = 程序在解析失败时自己编一个默认值；
 301:     · 这里 = 操作员在配置里**显式声明**"我们的件就放在这儿"，程序只负责读。
 302:     两者的区别是"谁做的决定"。读不到就返回 None，调用方继续 fail-closed。
 303: 
 304:     返回 None 的三种情况：文件不存在、没有 [PickPoint] 节、值解析不出来。
 305:     """
 306: 
 307:     p = Path(path)
 308:     if not p.exists():
 309:         return None
 310:     cfg = configparser.ConfigParser()
 311:     cfg.read(p, encoding="utf-8")
 312:     if not cfg.has_section("PickPoint"):
 313:         return None
 314:     try:
 315:         x_mm = cfg.getfloat("PickPoint", "x")
 316:         y_mm = cfg.getfloat("PickPoint", "y")
 317:     except (configparser.Error, ValueError):
 318:         return None
 319:     if not (math.isfinite(x_mm) and math.isfinite(y_mm)):
 320:         return None
 321:     return (int(round(x_mm * 10)), int(round(y_mm * 10)))

 325:     """从 sorting_config.ini 的 [DropPoints] 读六盒落点。
 326: 
 327:     文件里单位是 **mm**（`工具/control_panel.py` 按 mm 写），ITEM_TASK 要
 328:     **0.1mm**，这里换算。读不到条目就返回空表 —— 调用方 fail-closed，
 329:     绝不猜坐标：猜错 = 货落错盒，直接违反“形状颜色一致的放同一个盒”。
 330:     """
 331:     p = Path(path)
 332:     if not p.exists():
 333:         return {}
 334:     cfg = configparser.ConfigParser()
 335:     cfg.read(p, encoding="utf-8")
 336:     if not cfg.has_section("DropPoints"):
 337:         return {}
 338:     out: dict[int, tuple[int, int]] = {}
 339:     for key, value in cfg.items("DropPoints"):
 340:         try:
 341:             box_id = int(key)

 530: def build_frame_output(
 531:     kind: str, output_dir: str, logger: logging.Logger, *, demo: bool
 532: ):
 533:     """构造合成帧输出回调（注入 HdmiDisplayRenderer.frame_output）。
 534: 
 535:     生产模式（非 --fake-link）必须真的把帧送到屏上：取不到 /dev/fb0 就
 536:     SystemExit，**绝不静默降级为 none**。静默降级的后果是屏全黑但链路照跑、
 537:     DISPLAY_ACK 照发，直接撞赛规"15 秒没有显示本次分拣信息，本轮比赛结束"，
 538:     而且现场看不出任何异常。
 539:     """
 540: 
 541:     if kind == "auto":
 542:         if not demo:
 543:             raise SystemExit(
 544:                 "--frame-output auto 只用于 --fake-link 主机测试。\n"
 545:                 "  真实模式必须显式指定 --frame-output fb0（HDMI 屏）；\n"
 546:                 "  auto 会在取不到 /dev/fb0 时静默降级为 none —— 屏幕全黑却不报错。"
 547:             )
 548:         kind = "fb0" if os.path.exists("/dev/fb0") else "none"
 549:         logger.info("--frame-output auto -> %s（仅 --fake-link 主机测试）", kind)
 550: 
 551:     if kind == "fb0":
 552:         try:
 553:             # 启动期就试开一次：权限/设备缺失在此 fail-fast，而不是等第一件
 554:             # 物料走到渲染才炸（那时 STM32 已经动过了）。
 555:             with open("/dev/fb0", "rb"):
 556:                 pass
 557:         except OSError as exc:
 558:             raise SystemExit(
 559:                 f"打不开 /dev/fb0（{exc}）：拒绝启动。\n"
 560:                 "  检查：ls -l /dev/fb0；当前用户是否在 video 组；HDMI 是否已输出。\n"
 561:                 "  只想在主机上跑链路请加 --fake-link。"

 564:             # 按 fb 的真实 bpp 打包（16bpp→RGB565 小端，24/32bpp→BGR[X]）并处理行填充。
 565:             # 读不到真实 bpp 就 fail-closed —— 按 32bpp 猜的后果已实机踩过：
 566:             # 16bpp 的屏上写 1,843,200 字节 = [Errno 28] No space left on device。

 574:             ) from exc

 578:         directory.mkdir(parents=True, exist_ok=True)
 579:         counter = itertools.count(1)

 581:         def _to_file(img) -> None:

 590:                 "--frame-output none：本轮不会有任何屏幕输出。赛规要求本次分拣信息"

 592:             )
 593:         return lambda img: None

 643: def make_drop_point_lookup(
 644:     drop_points: dict[int, tuple[int, int]], logger: logging.Logger
 645: ) -> Callable[[int], tuple[int, int] | None]:
 646:     """安全落点查表：未知 box_id 返回 None，该件**拒识**，链路继续。
 647: 
 648:     这里**不能**直接传 `drop_points.__getitem__`：未知 box_id 抛 KeyError 会
 649:     一路冒到 run() 的兜底 `except`，把**整条链**带走（stopped，不是丢一件）。
 650:     返回 None 后由 build_item_task_queue 跳过该件 —— 不进队列、不发 ITEM_TASK。
 651: 
 652:     注意这**不是**"猜一个点"：拒识该件 ≠ 给一个默认坐标；落点表语义没动，
 653:     六盒坐标仍从 [DropPoints] 读。
 654:     """
 655: 
 656:     def lookup(box_id: int) -> tuple[int, int] | None:
 657:         point = drop_points.get(box_id)
 658:         if point is None:
 659:             logger.warning(
 660:                 "未知 box_id=%r：落点表只有 %s，拒识该件（不进队列、不发 ITEM_TASK）",
 661:                 box_id,
 662:                 sorted(drop_points),
 663:             )
 664:         return point
 665: 
 666:     return lookup

 668: def build_socket_source(
 669:     *,
 670:     round_id: str,
 671:     round_task_id: int,
 672:     drop_point_for_box: Callable[[int], tuple[int, int] | None],

 674:     fixed_pick_point: tuple[int, int] | None = None,

 676:     """真实视觉源：SocketVisionSource + **真实落点表** + **真实像素→毫米标定**。
 677: 
 678:     落点表来自 sorting_config.ini 的 [DropPoints]（见 load_drop_points），
 679:     由 `工具/control_panel.py` 现场标定后写入。查不到的 box_id 返回 None，
 680:     由 build_item_task_queue 跳过该件（拒识并记日志）—— 既不猜点，也不因为
 681:     一件查不到就把整条链带走。见 P11-A 设计 §1.3。
 682: 
 683:     抓取点来自 calibration_config.ini 的 [CameraCalibration]（像素→毫米仿射
 684:     + 高度视差补偿），由 `工具/calib_photo.py` 或 `工具/calib_affine.py` 产出。
 685: 
 686:     🔴 2026-09-18 接通（此前一直是断的）：
 687:     本函数原先**没有**传 grasp_point_for_target，于是 pick_x/pick_y 恒为 None，
 688:     `encode_item_task_payload` 抛 PayloadError（item_context.py:131），
 689:     **一条 ITEM_TASK 都发不出去、机器一步都不会动**。
 690:     件本身的 box_id 是有的（校赛模式把颜色折叠进 BATCH_TARGETS 的 id 字节，
 691:     见 vision_ingest.BATCH_ID_BYTE_IS_BOX_ID），所以缺的只有这一个。
 692: 
 693:     ⚠️ 未标定（参数缺失、仍是占位值全 0、或矩阵不可逆）时
 694:     `make_grasp_point_for_target` 一律返回 None → 该件被拒识 → 链路继续但不产出。
 695:     这是有意的 fail-closed：宁可不出货，也不要拿 (0,0) 把每一件都送到正中。
 696:     """
 697:     converter = PixelToDeviceConverter(calibration_config)
 698:     if converter.is_calibrated:
 699:         _LOGGER.info(
 700:             "抓取点来源 = 像素→毫米标定 %s（工作区 [%.1f,%.1f]×[%.1f,%.1f]mm，"
 701:             "物体高度 %d 类）",
 702:             converter.config_path,
 703:             converter.x_limits[0], converter.x_limits[1],
 704:             converter.y_limits[0], converter.y_limits[1],
 705:             len(converter.object_heights),
 706:         )
 707:         grasp_point_for_target = make_grasp_point_for_target(converter)
 708:     elif fixed_pick_point is not None:
 709:         # 固定摆放模式（2026-09-18 用户裁定）：件每次都放在同一个位置，
 710:         # 抓取点是常量，不需要像素→毫米标定。
 711:         _LOGGER.warning(
 712:             "标定未加载（%s）→ 改用【固定抓取点】模式：[PickPoint] = "
 713:             "(%.1f, %.1f)mm。**件必须每次放在同一位置**，"
 714:             "否则会抓空或抓偏。",
 715:             converter.config_path,
 716:             fixed_pick_point[0] / 10.0, fixed_pick_point[1] / 10.0,
 717:         )
 718: 
 719:         def grasp_point_for_target(target):
 720:             return fixed_pick_point
 721: 
 722:     else:
 723:         _LOGGER.error(
 724:             "抓取点来源缺失：标定未加载（%s）且 sorting_config.ini 没有 [PickPoint]。"
 725:             "本次运行**不会产出任何 ITEM_TASK**、机器不会动。"
 726:             "二选一：(a) 做像素→毫米标定填进 calibration_config.ini；"
 727:             "(b) 在 sorting_config.ini 写一节 [PickPoint] x=.. y=..（mm，件固定放一处）。",
 728:             converter.config_path,
 729:         )
 730:         # ⚠️ 这里必须给一个**总是返回 None 的回调**，不能真的传 None。
 731:         #    vision_ingest.py:240 的语义是：
 732:         #      回调为 None   → pick 留 None，该件**照样进队列**，
 733:         #                      直到 encode_item_task_payload 才抛 PayloadError；
 734:         #      回调返回 None → **当场拒识该件**（continue），不进队列。
 735:         #    我们要的是后者：没有抓取点来源时一帧就干脆不产出任何件，
 736:         #    而不是攒一个必然发不出去的 payload 再炸。
 737:         def grasp_point_for_target(target):
 738:             return None
 739: 
 740:     return SocketVisionSource(
 741:         round_id=round_id,
 742:         task_id=round_task_id,
 743:         image_id_for_class=_identity_image_id,
 744:         drop_point_for_box=drop_point_for_box,
 745:         grasp_point_for_target=grasp_point_for_target,

 899: # ---------------------------------------------------------------------------
 900: # 建轮次吞帧补偿（2026-09-18 实机定位）
 901: # ---------------------------------------------------------------------------
 902: def _should_resend_on_barrier_ready(
 903:     scheduler: Scheduler, last_task_context: ItemTaskContext | None
 904: ) -> bool:
 905:     """收到 BARRIER_READY 时，是否应当立刻原样重发在飞的那一帧 ITEM_TASK。
 906: 
 907:     背景（实机证据，不要凭直觉改）：
 908:       板子开机处于 FAULT_LOCK、round_id=0。收到第一发 ITEM_TASK 时它**只用这一发
 909:       建轮次**（ec_state_machine.c:76-86 → BEGIN_POST），然后自己跑
 910:       POST→HOMING→PREPARE_BARRIER→WAIT_TASK，到 WAIT_TASK 才 emit BARRIER_READY
 911:       （:127-131）。而建轮次那一发在 consume_and_emit 里**没有任何分支**
 912:       （ec_protocol_gateway.c:66-110），tx_len 保持 0 —— **一个字节都不回**。
 913:       同时 :79 清了 dedup、:85 直接 return，那个 identity 从未被记录，
 914:       所以**重发同一帧会被当成"全新首件"而正常 ACK**（04:13 实机复现：
 915:       第 1 发只回 BARRIER_READY，第 2 发同样字节回 ACK_ACCEPTED → 开始运动）。
 916: 
 917:       结论：BARRIER_READY 到达 = "刚建完轮次、那一发被启动序列吞了"，
 918:       此时若我们还在等 MOTION_DONE，就该立刻重发，而不是干等 5 秒超时。
 919: 
 920:     只在真的有一件在飞（WAIT_MOTION_DONE + 有 task 上下文）时重发。
 921:     板子每完成一件也会再报一次 BARRIER_READY（:203-209 → PREPARE_BARRIER），
 922:     那时 state 已不是 WAIT_MOTION_DONE，天然不会误触发。
 923:     """
 924: 
 925:     return last_task_context is not None and scheduler.state is SchedulerState.WAIT_MOTION_DONE
 926: 
 927: def _mm10(value: int | None) -> str:
 928:     """0.1mm 整数 → mm 字符串（None 原样显示）。仅用于日志/诊断。
 929: 
 930:     2026-09-18 实机踩过的坑：原日志只有 target_seq/class_id/image_id，没有坐标。
 931:     结果 X 轴被命令走到正方向机械极限顶死堵转，事后**完全看不出命令坐标是多少**，
 932:     只能靠猜。落点和抓取点都必须打出来。
 933:     """
 934:     return "None" if value is None else "%.1f" % (value / 10.0)

  91: @dataclass(frozen=True)
  92: class FramebufferFormat:
  93:     """fb 的真实几何（读自 sysfs，不是猜的）。
  94: 
  95:     line_length 是 fb 一行的字节数，可能大于 frame.width * bytes_per_pixel
  96:     （行填充）。为 None 表示内核没暴露 line_length/stride（本机 fkms 实测如此），
  97:     此时按紧凑行处理。
  98:     """
  99: 
 100:     bpp: int
 101:     virtual_width: int
 102:     virtual_height: int
 103:     line_length: int | None = None
 104: 
 105:     @property
 106:     def bytes_per_pixel(self) -> int:
 107:         try:
 108:             return _BYTES_PER_PIXEL[self.bpp]
 109:         except KeyError:
 110:             raise FramebufferFormatError(
 111:                 f"bits_per_pixel={self.bpp} 不支持（只支持 16/24/32）"
 112:             ) from None
 113: 
 114:     def __str__(self) -> str:
 115:         line = self.line_length if self.line_length else "读不到（按紧凑行）"
 116:         return (
 117:             f"{self.virtual_width}x{self.virtual_height} @{self.bpp}bpp"
 118:             f" line_length={line}"
 119:         )
 120: 
 121: 
 122: def _read_sysfs(path: str) -> str | None:
 123:     try:
 124:         with open(path) as handle:
 125:             return handle.read().strip()
 126:     except OSError:
 127:         return None
 128: 
 129: 
 130: def read_framebuffer_format(device: str = "/dev/fb0") -> FramebufferFormat:
 131:     """读 fb 的真实像素格式：bits_per_pixel / virtual_size / line_length（或 stride）。
 132: 
 133:     bpp 或 virtual_size 读不到就抛 FramebufferFormatError —— 不猜。sysfs 节点按
 134:     设备名（fb0）挂，与传入的路径无关。
 135:     """
 136: 
 137:     name = os.path.basename(device.rstrip("/")) or "fb0"
 138:     base = f"/sys/class/graphics/{name}"
 139: 
 140:     bpp_raw = _read_sysfs(f"{base}/bits_per_pixel")
 141:     if not (bpp_raw or "").isdigit():
 142:         raise FramebufferFormatError(
 143:             f"读不到 {base}/bits_per_pixel：拒绝猜 bpp。\n"
 144:             f"  查：ls /sys/class/graphics/ ; cat {base}/bits_per_pixel\n"
 145:             "  该节点缺失通常是 fbdev 兼容层没起来（vc4-kms-v3d 下 /dev/fb0 是影子节点）"
 146:         )
 147:     bpp = int(bpp_raw)
 148:     if bpp not in _BYTES_PER_PIXEL:
 149:         raise FramebufferFormatError(
 150:             f"bits_per_pixel={bpp} 不支持（只支持 16/24/32）："
 151:             "按错误的格式写屏就是花屏，拒绝写"
 152:         )
 153: 
 154:     virtual = _read_sysfs(f"{base}/virtual_size")
 155:     parts = (virtual or "").split(",")
 156:     if len(parts) < 2 or not all(part.strip().isdigit() for part in parts[:2]):
 157:         raise FramebufferFormatError(
 158:             f"读不到 {base}/virtual_size（读到 {virtual!r}）："
 159:             "没有它无法确认帧写不写得下，拒绝写"
 160:         )
 161: 
 162:     line_length = None
 163:     for node in ("line_length", "stride"):
 164:         raw = _read_sysfs(f"{base}/{node}")
 165:         if (raw or "").isdigit():
 166:             line_length = int(raw)
 167:             break
 168: 
 169:     return FramebufferFormat(
 170:         bpp=bpp,
 171:         virtual_width=int(parts[0]),
 172:         virtual_height=int(parts[1]),
 173:         line_length=line_length,
 174:     )

 177: def validate_frame_size(width: int, height: int, fmt: FramebufferFormat) -> None:
 178:     """帧尺寸 vs fb 可见区：写不下就报错；不完全一致只告警（会只覆盖左上角）。"""
 179: 
 180:     if width > fmt.virtual_width or height > fmt.virtual_height:
 181:         raise FramebufferFormatError(
 182:             f"帧 {width}x{height} 比 fb 可见区 {fmt.virtual_width}x{fmt.virtual_height} 大："
 183:             "写下去要么错位要么越界，拒绝写。\n"
 184:             "  分辨率由 config.txt 的 hdmi_cvt 与 --width/--height 决定，两边必须一致"
 185:         )
 186:     if (width, height) != (fmt.virtual_width, fmt.virtual_height):
 187:         logger.warning(
 188:             "帧 %dx%d ≠ fb 可见区 %s：只覆盖左上角，其余像素保持原样（不是花屏，是没画满）。"
 189:             "对齐办法：改 config.txt 的 hdmi_cvt，或让 --width/--height 与屏一致",
 190:             width,
 191:             height,
 192:             fmt,
 193:         )

 196: def encode_frame(frame: "PIL.Image.Image", fmt: FramebufferFormat) -> bytes:
 197:     """PIL RGB 帧 -> 可直接写 fb 的字节（按真实 bpp 打包 + 行填充）。
 198: 
 199:     · 16bpp：RGB565 小端 —— 红 (255,0,0) = 0xF800 = 字节 00 F8。
 200:       fbdev 的 fb 内容是 CPU 字节序的像素值，Pi 是小端；工具/pi_blit_png.py
 201:       用同一打包方式在本机 16bpp 上出图成功。
 202:     · 24/32bpp：fbdev truecolor 的内存序是 B,G,R[,X]
 203:       （red 在 bit16 / green 在 bit8 / blue 在 bit0）；32bpp 第 4 字节留 0。
 204:     · line_length 大于一行宽度（行填充）时，每行末尾补零，否则整幅会斜切。
 205:     """
 206: 
 207:     px = fmt.bytes_per_pixel
 208:     width, height = frame.size
 209:     validate_frame_size(width, height, fmt)
 210: 
 211:     row = width * px
 212:     stride = fmt.line_length or row
 213:     if stride < row:
 214:         raise FramebufferFormatError(
 215:             f"line_length/stride={stride} 小于一行应有的 {row} 字节"
 216:             f"（{width}px @{fmt.bpp}bpp）"
 217:         )
 218: 
 219:     rgb = frame.convert("RGB").tobytes()
 220:     out = bytearray(width * height * px)
 221:     index = 0
 222:     if fmt.bpp == 16:
 223:         for offset in range(0, len(rgb), 3):
 224:             value = (
 225:                 ((rgb[offset] & 0xF8) << 8)
 226:                 | ((rgb[offset + 1] & 0xFC) << 3)
 227:                 | (rgb[offset + 2] >> 3)
 228:             )
 229:             out[index] = value & 0xFF  # 小端：低字节先写
 230:             out[index + 1] = value >> 8
 231:             index += 2
 232:     else:  # 24/32bpp：B,G,R[,X]
 233:         for offset in range(0, len(rgb), 3):
 234:             out[index] = rgb[offset + 2]
 235:             out[index + 1] = rgb[offset + 1]
 236:             out[index + 2] = rgb[offset]
 237:             index += px  # 32bpp 的第 4 字节保持 0（屏不用）
 238: 
 239:     data = bytes(out)
 240:     if stride == row:
 241:         return data
 242:     pad = b"\x00" * (stride - row)
 243:     return b"".join(
 244:         data[at : at + row] + pad for at in range(0, len(data), row)
 245:     )

 267:     """Pillow-composes DisplayViewModel frames; implements the DisplayRenderer protocol.
 268: 
 269:     Constructed per item with the active ``ItemTaskContext`` so that
 270:     ``PendingRender.complete()`` rebuilds the exact ``DisplayContent``

 297:     def present(self, view: DisplayViewModel) -> PendingRender:
 298:         """Compose one frame from the view, output it, and open the render.
 299:

 301:         raises) or returns a non-image, ``ImageLoadError`` (a ValueError)

 310:             image_id=view.image_id,
 311:             success_count=view.success_count,
 312:             result_code=view.result_code,
 313:         )

 349:         draw.text(

 358:     def _load_cargo_image(self, view: DisplayViewModel, Image) -> "PIL.Image.Image":
 359:         try:
 360:             raw = self._image_loader(view.image_id)
 361:         except Exception as exc:
 362:             raise ImageLoadError(
 363:                 f"货物图片加载失败 image_id={view.image_id}: {exc!r}"
 364:             ) from exc
 365:         if not isinstance(raw, Image.Image):
 366:             raise ImageLoadError(
 367:                 f"image_loader(image_id={view.image_id}) 返回 {type(raw).__name__}，"
 368:                 "需要 PIL.Image.Image"
 369:             )
 370:         if raw.mode == "RGBA":
 371:             base = Image.new("RGB", raw.size, (255, 255, 255))
 372:             base.paste(raw, mask=raw.getchannel("A"))
 373:             return base
 374:         return raw.convert("RGB")

 376:     @staticmethod
 377:     def _fit_into(
 378:         img: "PIL.Image.Image", Image, box: tuple[int, int, int, int]
 379:     ) -> tuple["PIL.Image.Image", int, int]:
 380:         """Aspect-fit ``img`` into ``box``; return (fitted, paste_x, paste_y)."""
 381:         x0, y0, x1, y1 = box
 382:         box_w, box_h = x1 - x0, y1 - y0
 383:         scale = min(box_w / img.width, box_h / img.height)
 384:         new_w = max(1, round(img.width * scale))
 385:         new_h = max(1, round(img.height * scale))
 386:         # 2026-09-18 树莓派实机修复：Image.Resampling 是 Pillow 9.1 才有的，
 387:         # 而树莓派是 Pillow 8.1.2 → AttributeError: module 'PIL.Image' has no
 388:         # attribute 'Resampling'。getattr 兜底让 8.x 与 9.1+ 都能跑。
 389:         _resampling = getattr(Image, "Resampling", Image)
 390:         fitted = img.resize((new_w, new_h), _resampling.LANCZOS)
 391:         px = x0 + (box_w - new_w) // 2
 392:         py = y0 + (box_h - new_h) // 2
 393:         return fitted, px, py

 395:     @staticmethod
 396:     def _paste_clipped(
 397:         canvas: "PIL.Image.Image", fitted: "PIL.Image.Image", px: int, py: int
 398:     ) -> None:
 399:         """Paste ``fitted`` at (px, py), clipping to the canvas (custom sizes)."""
 400:         left = max(0, px)
 401:         top = max(0, py)
 402:         right = min(canvas.width, px + fitted.width)
 403:         bottom = min(canvas.height, py + fitted.height)
 404:         if left >= right or top >= bottom:
 405:             return
 406:         region = fitted.crop((left - px, top - py, right - px, bottom - py))
 407:         canvas.paste(region, (left, top))

  41: def validate_affine(a11: float, a12: float, tx: float,
  42:                     a21: float, a22: float, ty: float) -> Optional[str]:
  43:     """校验一组仿射参数；合法返回 None，不合法返回一句中文原因。
  44: 
  45:     🔴 为什么必须有这个（2026-09-18 实机前发现）：
  46:     ``calibration_config.ini`` 出厂带的是**占位值 a11..ty 全 0.0**。
  47:     旧代码只要 ``[CameraCalibration]`` 节存在就置 ``is_calibrated = True``，
  48:     于是 6 个参数全 0 也被当成"已标定"，``pixel_to_device()`` 不再返回 None，
  49:     而是把**每一个物体**都算成 (0, 0) —— 全部去装置正中抓。
  50:     这比"未标定"危险得多：未标定是 fail-closed（拒识全部），
  51:     假标定是 fail-open（静默地全抓错）。
  52:     """
  53: 
  54:     values = (a11, a12, tx, a21, a22, ty)
  55:     if any(not math.isfinite(v) for v in values):
  56:         return "参数含 NaN/Inf"
  57: 
  58:     if all(abs(v) < 1e-12 for v in values):
  59:         return ("6 个参数全是 0.0 —— 这是配置文件里的占位值，还没做标定。"
  60:                 "全 0 会把每个物体都算成装置原点 (0,0)，全部去正中抓")
  61: 
  62:     det = a11 * a22 - a12 * a21
  63:     scale_x = math.hypot(a11, a12)
  64:     scale_y = math.hypot(a21, a22)
  65:     # 用【相对】判据：行列式的合理下限与两个尺度之积同阶。
  66:     # 纯绝对阈值有两个毛病：会误杀"数值大但形状良态"的合法标定，
  67:     # 又会放过 `a11=1,a12=1,a21=1,a22=1.000000000001` 这种
  68:     # "看着全是正常数、其实把平面压成一条线"的退化矩阵。
  69:     if abs(det) < 1e-9 * scale_x * scale_y:
  70:         return (f"2x2 子阵接近奇异（det={det:.3e}，尺度积="
  71:                 f"{scale_x * scale_y:.3e}）：像素平面几乎被压成一条线，"
  72:                 "解不出唯一位置")
  73: 
  74:     # 量级检查：a11/a12 是"每像素多少毫米"。320 像素要看住约 200mm 的台面，
  75:     # 大致在 0.1~10 mm/px。给一个宽松但能拦住明显垃圾的区间。
  76:     for name, scale in (("X", scale_x), ("Y", scale_y)):
  77:         if not (0.02 <= scale <= 20.0):
  78:             return (f"{name} 方向每像素 {scale:.4f} mm，超出合理范围 0.02~20 —— "
  79:                     "量级不对，多半是参数填错或单位混了")
  80: 
  81:     # 平移量检查：仿射的输出就是台面毫米，量级和台面同阶。
  82:     # `ty = 1e9` 这种"读得出来但显然荒谬"的值必须拦住。
  83:     for name, value in (("tx", tx), ("ty", ty)):
  84:         if abs(value) > 1.0e4:
  85:             return (f"{name}={value:g} 超出合理量级（|{name}| > 1e4 mm）—— "
  86:                     "多半是参数填错，或 mm 与 0.1mm 混了")
  87: 
  88:     return None
  89: 
  90: 
  91: # ITEM_TASK 的 X/Y 是协议可序列化的 ±145.0mm（interface-contract §5.2，
  92: # 镜像固件 EC_MOTION_WORKSPACE_LIMIT_MM10 = 1450）。
  93: # 工作区限制必须落在它里面：否则本模块会"放行"一个直到 encode 才炸的坐标，
  94: # 那件会在发帧那一刻把整条链带走，而不是安静地被拒识。
  95: _SERIALIZABLE_LIMIT_MM = 145.0
  96: 
  97: 
  98: def validate_workspace(x_limits: Tuple[float, float],
  99:                        y_limits: Tuple[float, float]) -> Optional[str]:
 100:     """校验工作区限制；合法返回 None，否则返回一句中文原因。
 101: 
 102:     三条：有限；min <= max；落在 ±145.0mm 内。
 103:     """
 104: 
 105:     for name, limits in (("X", x_limits), ("Y", y_limits)):
 106:         lo, hi = limits
 107:         if not (math.isfinite(lo) and math.isfinite(hi)):
 108:             return f"{name} 限制含 inf/NaN（{lo}, {hi}）"
 109:         if lo > hi:
 110:             return f"{name} 的 min={lo} 大于 max={hi}"
 111:         if lo < -_SERIALIZABLE_LIMIT_MM or hi > _SERIALIZABLE_LIMIT_MM:
 112:             return (f"{name} 限制 [{lo}, {hi}] 超出协议可序列化范围 "
 113:                     f"±{_SERIALIZABLE_LIMIT_MM}mm —— 越界坐标会在 "
 114:                     "encode_item_task_payload 那一刻才抛，会把那件乃至整条链带走")
 115:     return None

 153:                     e, self.config_path

 155:                 self.is_calibrated = False
 156:         else:
 157:             _LOGGER.warning(

 162:     def _load_calibration(self) -> None:
 163:         """从配置文件加载标定参数（内部方法，构造时调用）"""
 164:         # 🔴 必须允许行内注释：calibration_config.ini 的 [ObjectHeights] 每行都写成
 165:         #    `1 = 40.0  # 青色 cyan 正方体`。configparser 默认不剥行内注释，
 166:         #    float("40.0  # 青色 ...") 会抛 ValueError → 每一项都被静默跳过
 167:         #    → object_heights 全空 → 每个件都查不到高度。
 168:         config = configparser.ConfigParser(inline_comment_prefixes=("#",))
 169:         config.read(self.config_path, encoding='utf-8')
 170: 
 171:         if 'CameraCalibration' not in config:
 172:             raise ValueError(f"缺少 [CameraCalibration] 节: {self.config_path}")
 173: 
 174:         cal = config['CameraCalibration']
 175: 
 176:         # 读取仿射矩阵（6 参数 → 3x3 矩阵）
 177:         # x = a11*u + a12*v + tx
 178:         # y = a21*u + a22*v + ty
 179:         affine = (cal.getfloat('a11'), cal.getfloat('a12'), cal.getfloat('tx'),
 180:                   cal.getfloat('a21'), cal.getfloat('a22'), cal.getfloat('ty'))
 181:         # 🔴 必须先校验再落盘：占位值全 0 也"能读出来"，但用它换算会把
 182:         #    每个物体都算成 (0,0)。见 validate_affine 的说明。
 183:         problem = validate_affine(*affine)
 184:         if problem is not None:
 185:             raise ValueError(f"仿射参数不合法：{problem}")
 186: 
 187:         self.affine_matrix = np.array([
 188:             [affine[0], affine[1], affine[2]],
 189:             [affine[3], affine[4], affine[5]],
 190:             [0.0, 0.0, 1.0]
 191:         ], dtype=np.float64)
 192: 
 193:         self.camera_height = cal.getfloat('camera_height')
 194:         if not (self.camera_height > 0.0 and math.isfinite(self.camera_height)):
 195:             raise ValueError(f"camera_height={self.camera_height} 不是正数")
 196: 
 197:         # 帧尺寸只用于定位"图像中心"（光轴落点）。MaixCAM 当前是 320x320。
 198:         self.frame_width = cal.getfloat('frame_width', fallback=320.0)
 199:         self.frame_height = cal.getfloat('frame_height', fallback=320.0)
 200:         if not (math.isfinite(self.frame_width) and self.frame_width > 0.0
 201:                 and math.isfinite(self.frame_height) and self.frame_height > 0.0):
 202:             raise ValueError(
 203:                 f"frame_width/height={self.frame_width}/{self.frame_height} "
 204:                 "必须是有限正数（inf/NaN 会让光轴落点算不出来）"
 205:             )
 206: 
 207:         # 读取物体高度（可选节）
 208:         if 'ObjectHeights' in config:
 209:             for box_id_str, height_str in config['ObjectHeights'].items():
 210:                 try:
 211:                     box_id = int(box_id_str)
 212:                     height = float(height_str)
 213:                     self.object_heights[box_id] = height
 214:                 except ValueError:
 215:                     _LOGGER.warning(
 216:                         "跳过无效物体高度配置: %s = %s",
 217:                         box_id_str, height_str
 218:                     )
 219: 
 220:         # 读取工作区限制（可选节）
 221:         if 'WorkspaceLimits' in config:
 222:             lim = config['WorkspaceLimits']
 223:             self.x_limits = (lim.getfloat('x_min'), lim.getfloat('x_max'))
 224:             self.y_limits = (lim.getfloat('y_min'), lim.getfloat('y_max'))
 225:             problem = validate_workspace(self.x_limits, self.y_limits)
 226:             if problem is not None:
 227:                 raise ValueError(f"工作区限制不合法：{problem}")
 228: 
 229:         self.is_calibrated = True
 230:         _LOGGER.info(
 231:             "标定配置已加载: camera_height=%.1fmm, 物体高度 %d 类, 工作区 [%.1f,%.1f]×[%.1f,%.1f]mm",
 232:             self.camera_height,
 233:             len(self.object_heights),
 234:             self.x_limits[0], self.x_limits[1],
 235:             self.y_limits[0], self.y_limits[1]
 236:         )

 238:     def _optical_axis_point(self) -> Tuple[float, float]:
 239:         """光轴落点（nadir）在**装置坐标**里的位置 = 图像中心经仿射变换后的像。
 240: 
 241:         🔴 为什么必须有这个（2026-09-18 实机前发现）：
 242:         「高度视差」是物体越高、成像越向**远离光轴**的方向扩张 ——
 243:         张开中心是**光轴落点**，不是装置原点。
 244:         旧写法 ``x_bottom = x_top * scale`` 是"以装置原点为中心缩放"，
 245:         只有当相机恰好装在装置正上方（光轴落点 = 原点）时才等价。
 246: 
 247:         本项目相机**明确不在正中间**（用户 2026-09-18 确认），所以旧写法会引入
 248:         固定偏差 ``nadir_x * (1 - scale)``：
 249:             nadir 偏 30mm、三棱柱 h=35mm(d=262mm, scale≈1.154) → 偏差 −4.6mm
 250:             nadir 偏 50mm                                          → 偏差 −7.7mm
 251:         而抓取误差预算只有 5mm。
 252: 
 253:         假设：主点 ≈ 图像中心。若日后发现主点明显偏移，把
 254:         ``frame_width/frame_height`` 换成能表达主点位置的口径，或在配置里
 255:         另行给出 —— 不要在这里悄悄塞一个常数。
 256:         """
 257: 
 258:         cx = self.frame_width / 2.0
 259:         cy = self.frame_height / 2.0
 260:         matrix = self.affine_matrix
 261:         return (float(matrix[0][0] * cx + matrix[0][1] * cy + matrix[0][2]),
 262:                 float(matrix[1][0] * cx + matrix[1][1] * cy + matrix[1][2]))

 264:     def pixel_to_device(
 265:         self,
 266:         u: float,
 267:         v: float,
 268:         class_id: int
 269:     ) -> Optional[Tuple[int, int]]:
 270:         """像素坐标 → 装置坐标（带视差补偿）
 271: 
 272:         Args:
 273:             u, v: 像素坐标（bbox 中心）
 274:             class_id: 历史参数名，保留关键字兼容；实际传 [ObjectHeights] 的
 275:                       box_id（颜色折叠后的货格号），不是 VisionTarget.class_id。
 276: 
 277:         Returns:
 278:             (x_mm10, y_mm10): 装置坐标，单位 0.1mm，原点=装置中心
 279:             None: 未标定 / 超出工作区 / 像素或高度非法 / 高度缺失
 280: 
 281:         ⚠️ 超出工作区的物体应被拒识（记日志，不发 ITEM_TASK）
 282:         """
 283:         if not self.is_calibrated:
 284:             return None
 285: 
 286:         # Step 0: 输入校验 —— 非法像素一律返回 None，不抛异常。
 287:         # 视觉侧给来的 (u,v) 直接来自 wire，宁可拒识这一件，也不能让
 288:         # 一个 NaN/字符串炸在换算里（那会把整条链带走）。
 289:         for name, value in (("u", u), ("v", v)):
 290:             if isinstance(value, bool) or not isinstance(value, (int, float)):
 291:                 _LOGGER.warning("像素 %s=%r 不是数字：拒识该件", name, value)
 292:                 return None
 293:             if not math.isfinite(float(value)):
 294:                 _LOGGER.warning("像素 %s=%r 不是有限数：拒识该件", name, value)
 295:                 return None
 296: 
 297:         # 物体高度必须**在配置里明确给出**。
 298:         # 🔴 2026-09-18 收紧：以前查不到就按 h=0（无视差补偿）静默继续 ——
 299:         #    那等于"配置漏了一类，就悄悄少算一截视差"，误差直接进抓取点。
 300:         #    现在查不到 → 拒识该件（fail-closed），并说清是哪个 id。
 301:         try:
 302:             h = self.object_heights[class_id]
 303:         except KeyError:
 304:             _LOGGER.warning(
 305:                 "object_id=%r 不在 [ObjectHeights]（已配置 %s）：拒识该件。"
 306:                 "不要用 h=0 顶替 —— 那会静默少算视差",
 307:                 class_id, sorted(self.object_heights),
 308:             )
 309:             return None
 310:         if not (math.isfinite(h) and 0.0 <= h < self.camera_height):
 311:             _LOGGER.error(
 312:                 "object_id=%r 的高度 h=%r 非法（要求 0 <= h < camera_height=%.1f）："
 313:                 "拒识该件，绝不当成 h=0 静默继续",
 314:                 class_id, h, self.camera_height,
 315:             )
 316:             return None
 317: 
 318:         # Step 1: 仿射变换（像素 → 顶面装置坐标 mm）
 319:         pixel_vec = np.array([float(u), float(v), 1.0], dtype=np.float64)
 320:         device_vec = self.affine_matrix @ pixel_vec
 321:         x_top, y_top = float(device_vec[0]), float(device_vec[1])
 322: 
 323:         # Step 2: 高度视差补偿 —— 以【镜头光轴落点】为基准，不是装置原点
 324:         if h > 0.0:
 325:             # 🔴 2026-09-18 修正：这里必须是【除】，不是乘。
 326:             #   相机在高度 H、光轴落 N。物体底面 B、高 h，其【顶面】在画面里
 327:             #   看起来"跑"到了更远的地面位置
 328:             #       G = N + (B - N) × H/(H-h)          （scale = H/(H-h) > 1）
 329:             #   仿射是在【台面】上标定的，所以 affine(像素) 给的是 G（表观位置），
 330:             #   不是 B。要还原真实底面：
 331:             #       B = N + (G - N) × (H-h)/H = N + (G - N) ÷ scale
 332:             #   具体算例（H=262, h=40, scale=1.1802, 光轴在原点, B=(0,50)）：
 333:             #       射线落到地面 = 50×1.1802 = 59.0 = G
 334:             #       59.0 ÷ 1.1802 = 50.0 = B  ✓
 335:             #       59.0 × 1.1802 = 69.6      ✗（旧实现，误差还翻倍）
 336:             #   光轴上的点 G=N 两种写法都得 N，所以「保持光轴不变」用旧写法也能过 ——
 337:             #   这正是这个 bug 一直没被发现的原因。
 338:             scale = self.camera_height / (self.camera_height - h)
 339:             nadir_x, nadir_y = self._optical_axis_point()
 340:             x_bottom = nadir_x + (x_top - nadir_x) / scale
 341:             y_bottom = nadir_y + (y_top - nadir_y) / scale
 342:         else:
 343:             x_bottom, y_bottom = x_top, y_top
 344: 
 345:         # Step 3: 边界检查（用**补偿后**的坐标，不是补偿前的）
 346:         if not (self.x_limits[0] <= x_bottom <= self.x_limits[1]):
 347:             return None  # 超出 X 范围
 348:         if not (self.y_limits[0] <= y_bottom <= self.y_limits[1]):
 349:             return None  # 超出 Y 范围
 350: 
 351:         # Step 4: 转换为 0.1mm 单位（协议单位）
 352:         x_mm10 = int(round(x_bottom * 10))
 353:         y_mm10 = int(round(y_bottom * 10))
 354: 
 355:         return (x_mm10, y_mm10)
 356: 
 357: def make_grasp_point_for_target(converter: PixelToDeviceConverter) -> GraspPointForTarget:
 358:     """造出 ``build_item_task_queue`` 要的 ``grasp_point_for_target``。
 359: 
 360:     入参是 ``VisionTarget``，出参是 ``(x_mm10, y_mm10)`` 或 ``None``。
 361: 
 362:     ⚠️ 用 ``box_id`` 查物体高度，**不是** ``class_id``：
 363:     ``[ObjectHeights]`` 的键是 box_id（见 ``calibration_config.ini`` 的注释）。
 364:     ``VisionTarget.class_id`` 的形状语义与 ``box_id`` 不等价。
 365:     当前校赛解析器把同一 id 字节同时写进两字段，但既有逻辑测试允许两者不同；
 366:     高度查询只能依照明确的 box_id，不能依赖 wire 解析产生的数值相等。
 367: 
 368:     未标定 / 越界 / box_id 缺失 一律返回 ``None``，该件被 ``build_item_task_queue``
 369:     丢件（拒识），**绝不**退回 (0,0) —— 那就是"所有件都去正中抓"，等于没有视觉。
 370:     """
 371: 
 372:     def lookup(target: VisionTarget) -> Optional[Tuple[int, int]]:
 373:         if not converter.is_calibrated:
 374:             return None
 375:         box_id = target.box_id
 376:         if box_id is None:
 377:             return None
 378:         return converter.pixel_to_device(target.pixel_x, target.pixel_y, box_id)
 379: 
 380:     return lookup
