 276: class StableTargets:
 277:     """每色一件：连续确认、时间平滑、短时漏检保留灰框；缓存不输出货格。"""
 278:     def __init__(self):

 281:     def update(self, detections, now):
 282:         seen = set()
 283:         for row in detections:
 284:             color = row["color"]
 285:             seen.add(color)
 286:             box = np.array(row["bbox"], dtype=float)
 287:             previous = self.tracks.get(color)
 288:             new_target = previous is None or now-previous["last_seen"] > MAX_TRACK_GAP
 289:             if not new_target:
 290:                 old = previous["box"]
 291:                 distance = float(np.linalg.norm((box[:2]+box[2:]/2)-(old[:2]+old[2:]/2)))
 292:                 new_target = distance > max(30.0, 0.9*max(old[2], old[3]))
 293:             if new_target:
 294:                 previous = {"box": box, "hits": 0, "confirmed": False,
 295:                             "last_seen": now, "row": row}
 296:                 self.tracks[color] = previous
 297:             else:
 298:                 dt = max(0.0, now-previous["last_seen"])
 299:                 alpha = 1.0 - np.exp(-dt / SMOOTH_TIME)
 300:                 previous["box"] += alpha*(box-previous["box"])
 301:             previous["hits"] += 1
 302:             previous["confirmed"] = previous["confirmed"] or previous["hits"] >= CONFIRM_FRAMES
 303:             previous["last_seen"], previous["row"] = now, row
 304:         results = []
 305:         for color, track in list(self.tracks.items()):
 306:             age = now-track["last_seen"]
 307:             if color not in seen:
 308:                 track["hits"] = 0
 309:                 if age > HOLD_SECONDS or not track["confirmed"]:
 310:                     del self.tracks[color]
 311:                     continue
 312:             if not track["confirmed"]:
 313:                 continue
 314:             row = dict(track["row"])
 315:             row["bbox"] = tuple(int(round(v)) for v in track["box"])
 316:             row["stale"] = color not in seen
 317:             if row["stale"]:
 318:                 row["bin"], row["classification_source"], row["reason"] = None, "none", "short_hold"
 319:             results.append(row)
 320:         results.sort(key=lambda r: (r["bbox"][1], r["bbox"][0]))
 321:         return results
