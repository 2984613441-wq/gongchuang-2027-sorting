### 模块1：插补算法模块（8h）

**新建文件**：`电控/firmware/stm32f407_m144z/src/interpolation.c`

```c
#include "interpolation.h"
#include <stdlib.h>

void interp_init(interp_state_t *state, 
                 int32_t x0, int32_t y0, 
                 int32_t x1, int32_t y1) {
    state->x0 = x0;
    state->y0 = y0;
    state->x1 = x1;
    state->y1 = y1;
    state->x_current = x0;
    state->y_current = y0;
    
    state->dx = abs(x1 - x0);
    state->dy = abs(y1 - y0);
    
    state->sx = (x0 < x1) ? 1 : -1;
    state->sy = (y0 < y1) ? 1 : -1;
    
    state->error = state->dx - state->dy;
    state->finished = false;
}

interp_axis_t interp_step(interp_state_t *state) {
    if (state->finished) {
        return INTERP_AXIS_NONE;
    }
    
    // 到达目标
    if (state->x_current == state->x1 && state->y_current == state->y1) {
        state->finished = true;
        return INTERP_AXIS_NONE;
    }
    
    // Bresenham算法
    int32_t e2 = 2 * state->error;
    interp_axis_t axis_to_move = INTERP_AXIS_NONE;
    
    if (e2 > -state->dy && state->x_current != state->x1) {
        state->error -= state->dy;
        state->x_current += state->sx;
        axis_to_move |= INTERP_AXIS_X;
    }
    
    if (e2 < state->dx && state->y_current != state->y1) {
        state->error += state->dx;
        state->y_current += state->sy;
        axis_to_move |= INTERP_AXIS_Y;
    }
    
    return axis_to_move;
}

bool interp_is_finished(const interp_state_t *state) {
    return state->finished;
}
```

**头文件**：`电控/firmware/stm32f407_m144z/src/interpolation.h`

```c
#ifndef INTERPOLATION_H
#define INTERPOLATION_H

#include <stdint.h>
#include <stdbool.h>

typedef enum {
    INTERP_AXIS_NONE = 0,
    INTERP_AXIS_X = 1,
    INTERP_AXIS_Y = 2,
    INTERP_AXIS_BOTH = 3
} interp_axis_t;

typedef struct {
    int32_t x0, y0;
    int32_t x1, y1;
    int32_t x_current;
    int32_t y_current;
    int32_t dx, dy;
    int32_t sx, sy;
    int32_t error;
    bool finished;
} interp_state_t;

void interp_init(interp_state_t *state, int32_t x0, int32_t y0, int32_t x1, int32_t y1);
interp_axis_t interp_step(interp_state_t *state);
bool interp_is_finished(const interp_state_t *state);

#endif

    MOTION_STATE_IDLE,

    interp_state_t interp;
    uint32_t step_timer_hz;
    uint32_t next_step_tick;
} motion_controller_t;

static motion_controller_t g_motion = {0};
```

**新增函数**：
```c
void motion_start_xy(int32_t target_x, int32_t target_y) {
    int32_t current_x = axis_get_position(AXIS_X);
    int32_t current_y = axis_get_position(AXIS_Y);
    
    interp_init(&g_motion.interp, current_x, current_y, target_x, target_y);
    
    g_motion.step_timer_hz = 1000;
    g_motion.next_step_tick = HAL_GetTick();
    g_motion.state = MOTION_STATE_XY_INTERP;
    
    axis_enable(AXIS_X);
    axis_enable(AXIS_Y);
}

void motion_poll(void) {
    if (g_motion.state != MOTION_STATE_XY_INTERP) {
        return;
    }
    
    uint32_t now = HAL_GetTick();
    
    if (now < g_motion.next_step_tick) {
        return;
    }
    
    // 插补一步
    interp_axis_t axis = interp_step(&g_motion.interp);
    
    if (axis == INTERP_AXIS_NONE) {
        g_motion.state = MOTION_STATE_DONE;
        return;
    }
    
    // 驱动电机
    if (axis & INTERP_AXIS_X) {
        axis_step_once(AXIS_X);
    }
    
    if (axis & INTERP_AXIS_Y) {
        axis_step_once(AXIS_Y);
    }
    
    g_motion.next_step_tick = now + (1000 / g_motion.step_timer_hz);
}
```

bool motion_check_limit_xy(void) {
    if (axis_limit_triggered(AXIS_X)) {
        axis_stop(AXIS_X);
        axis_stop(AXIS_Y);
        return false;
    }
    
    if (axis_limit_triggered(AXIS_Y)) {
        axis_stop(AXIS_X);
        axis_stop(AXIS_Y);
        return false;
    }
    
    return true;
}

#include "interpolation.h"
#include "test_support.h"

void test_interpolation(void) {
    interp_state_t state;
    
    // 测试1：45度斜线
    interp_init(&state, 0, 0, 10, 10);
    int x_steps = 0, y_steps = 0;
    
    while (!interp_is_finished(&state)) {
        interp_axis_t axis = interp_step(&state);
        if (axis & INTERP_AXIS_X) x_steps++;
        if (axis & INTERP_AXIS_Y) y_steps++;
    }
    
    TEST_ASSERT(x_steps == 10);
    TEST_ASSERT(y_steps == 10);
    TEST_ASSERT(state.x_current == 10);
    TEST_ASSERT(state.y_current == 10);
    
    // 测试2：浅角度
    interp_init(&state, 0, 0, 100, 20);
    x_steps = y_steps = 0;
    
    while (!interp_is_finished(&state)) {
        interp_axis_t axis = interp_step(&state);
        if (axis & INTERP_AXIS_X) x_steps++;
        if (axis & INTERP_AXIS_Y) y_steps++;
    }
    
    TEST_ASSERT(x_steps == 100);
    TEST_ASSERT(y_steps == 20);
}
