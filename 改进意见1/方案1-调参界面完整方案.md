#### Step 2: 参数同步脚本（2h）

**文件位置**：`/home/pi/sync_params.py`

```python
#!/usr/bin/env python3
import configparser
import serial
import socket
import struct
import time

def load_config():
    config = configparser.ConfigParser()
    config.read('/home/pi/sorting_config.ini')
    return config

def sync_to_stm32(config, port='/dev/ttyAMA0', baudrate=115200):
    """通过UART发送参数更新到STM32"""
    ser = serial.Serial(port, baudrate, timeout=1)
    
    # 构造CONFIG_UPDATE帧（命令字0x10）
    x_speed = int(config.getfloat('Motion', 'x_speed_multiplier') * 1000)
    y_speed = int(config.getfloat('Motion', 'y_speed_multiplier') * 1000)
    
    # 帧格式：0xA5 0x10 [len] [x_speed_2B] [y_speed_2B] [checksum]
    payload = struct.pack('>HH', x_speed, y_speed)
    frame = bytes([0xA5, 0x10, len(payload)]) + payload
    checksum = sum(frame) & 0xFF
    frame += bytes([checksum])
    
    ser.write(frame)
    
    # 等待STM32 ACK
    ack = ser.read(5)
    if ack and ack[0] == 0xA5 and ack[1] == 0x11:  # ACK命令字
        print("✅ STM32参数同步成功")
        return True
    else:
        print("❌ STM32参数同步失败")
        return False
    
def sync_to_maixcam(config, host='172.20.10.2', port=8765):
    """通过TCP发送参数更新到MaixCAM"""
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect((host, port))
    
    conf_th = config.getfloat('Vision', 'confidence_threshold')
    iou_th = config.getfloat('Vision', 'iou_threshold')
    
    # JSON格式参数更新
    import json
    cmd = {
        'type': 'UPDATE_PARAMS',
        'conf_th': conf_th,
        'iou_th': iou_th
    }
    sock.sendall(json.dumps(cmd).encode())
    
    # 等待ACK
    ack = sock.recv(128)
    if b'OK' in ack:
        print("✅ MaixCAM参数同步成功")
        return True
    else:
        print("❌ MaixCAM参数同步失败")
        return False

if __name__ == '__main__':
    config = load_config()
    
    print("开始同步参数...")
    stm32_ok = sync_to_stm32(config)
    maixcam_ok = sync_to_maixcam(config)
    
    if stm32_ok and maixcam_ok:
        print("\n✅ 所有参数同步完成！")
    else:
        print("\n⚠️ 部分参数同步失败，请检查日志")
```

#### Step 3: 修改STM32固件接收参数（1h）

**文件位置**：`电控/firmware/stm32f407_m144z/src/config_handler.c`

```c
// 新增CONFIG_UPDATE命令处理
void handle_config_update(const uint8_t *payload, uint16_t len) {
    if (len < 4) {
        send_nack(0x10, ERROR_INVALID_PAYLOAD);
        return;
    }
    
    // 解析参数
    uint16_t x_speed_multiplier = (payload[0] << 8) | payload[1];
    uint16_t y_speed_multiplier = (payload[2] << 8) | payload[3];
    
    // 仅在IDLE状态允许参数修改
    if (g_state != STATE_IDLE) {
        send_nack(0x10, ERROR_STATE_LOCKED);
        return;
    }
    
    // 应用参数
    motion_set_speed_multiplier(AXIS_X, x_speed_multiplier / 1000.0f);
    motion_set_speed_multiplier(AXIS_Y, y_speed_multiplier / 1000.0f);
    
    // 持久化到EEPROM
    eeprom_write_config();
    
    // 发送ACK
    send_ack(0x10);
}
```

# 在循环前增加参数管理器
class ParamManager:
    def __init__(self):
        self.conf_th = CONF_TH
        self.iou_th = IOU_TH
        self.lock = False
    
    def try_update(self, sock):
        """非阻塞接收参数更新"""
        if self.lock:
            return
        try:
            sock.settimeout(0.001)
            data = sock.recv(128)
            if data:
                import json
                cmd = json.loads(data)
                if cmd.get('type') == 'UPDATE_PARAMS':
                    self.conf_th = cmd['conf_th']
                    self.iou_th = cmd['iou_th']
                    sock.sendall(b'OK')
                    print(f"[PARAM] updated: conf={self.conf_th}, iou={self.iou_th}")
        except:
            pass

params = ParamManager()

# 主循环中
while not app.need_exit():

**风险1**：UART通信失败
- 缓解：预先测试串口，准备降级方案（手动烧录）

**风险2**：参数格式错误导致系统崩溃
- 缓解：增加参数校验，范围检查

**风险3**：MaixCAM网络断连
- 缓解：参数更新失败不影响已有参数，可继续运行
