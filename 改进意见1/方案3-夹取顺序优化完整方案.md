# 问题3：夹取顺序优化 - 完整实施方案

## 📋 问题本质

在准确性和时间之间寻找最优路径，根据视觉置信度、物块位置、夹取难度和已验证成功率动态排序，在15秒边界内最大化完成数量和准确性。

## ⚠️ 关键约束

1. **赛规红线**：不允许自动跳件（任何恢复都禁止自动跳件）
2. **15秒判罚**：15秒无动作本轮结束
3. **架构约束**：顺序决策在树莓派，STM32只执行单件任务

---

## 🎯 推荐方案：时间分段风险策略

### 核心思路

将15秒分为三个阶段，每个阶段采用不同的决策策略：

```
阶段1（0-7s）：快速收割期
  → 高置信度 + 近距离物块
  → 目标：完成3-4件

阶段2（7-12s）：稳定填充期
  → TSP路径优化
  → 只选Safe物块

阶段3（12-15s）：保守收尾期
  → confidence>0.9 且 time<1.5s
  → 预留3s安全边界
```

### Step 1: 物块风险分级系统（2h）

**文件位置**：`电控/src/electrical_control/scheduler/risk_classifier.py`

```python
from dataclasses import dataclass
from enum import Enum
from typing import List

class RiskLevel(Enum):
    SAFE = "safe"         # confidence>0.85 & success_rate>0.8
    MEDIUM = "medium"     # confidence>0.7
    RISKY = "risky"       # 其他

@dataclass
class ObjectRisk:
    object_id: int
    class_id: int
    confidence: float
    position: tuple[int, int]  # (x, y)
    estimated_time_s: float
    historical_success_rate: float
    risk_level: RiskLevel
    failure_count: int = 0

class RiskClassifier:
    def __init__(self):
        # 历史成功率字典：{class_id: success_rate}
        self.class_success_rates = {
            0: 0.9,  # 初始先验
            1: 0.85,
            2: 0.8,
            # ... 其他类别
        }
    
    def classify(self, obj: dict) -> ObjectRisk:
        """对单个物块进行风险分级"""
        class_id = obj['class_id']
        confidence = obj['confidence']
        position = (obj['x'], obj['y'])
        
        # 估算时间：距离 + 夹取动作
        distance = self._calc_distance(position)
        estimated_time = distance / 50.0 + 2.0  # 假设速度50mm/s + 2s夹取
        
        # 获取历史成功率
        success_rate = self.class_success_rates.get(class_id, 0.7)
        
        # 风险分级
        if confidence > 0.85 and success_rate > 0.8:
            risk_level = RiskLevel.SAFE
        elif confidence > 0.7:
            risk_level = RiskLevel.MEDIUM
        else:
            risk_level = RiskLevel.RISKY
        
        return ObjectRisk(
            object_id=obj['id'],
            class_id=class_id,
            confidence=confidence,
            position=position,
            estimated_time_s=estimated_time,
            historical_success_rate=success_rate,
            risk_level=risk_level
        )
    
    def update_success_rate(self, class_id: int, success: bool):
        """贝叶斯更新成功率"""
        current = self.class_success_rates.get(class_id, 0.7)
        if success:
            new_rate = 0.9 * current + 0.1
        else:
            new_rate = 0.7 * current
        self.class_success_rates[class_id] = max(0.1, min(0.99, new_rate))
    
    def _calc_distance(self, pos: tuple[int, int]) -> float:
        """计算当前位置到目标的欧氏距离（mm）"""
        # TODO: 从电机状态获取当前位置
        current_x, current_y = 0, 0
        return ((pos[0] - current_x)**2 + (pos[1] - current_y)**2)**0.5
```

### Step 2: 三阶段调度器（4h）

**文件位置**：`电控/src/electrical_control/scheduler/phased_scheduler.py`

```python
import time
from typing import List, Optional
from .risk_classifier import RiskClassifier, RiskLevel, ObjectRisk

class PhasedScheduler:
    def __init__(self, time_limit_s: float = 15.0):
        self.time_limit = time_limit_s
        self.start_time = None
        self.classifier = RiskClassifier()
        self.objects: List[ObjectRisk] = []
        self.completed_ids = set()
    
    def start_round(self, objects: List[dict]):
        """开始新一轮，初始化物块列表"""
        self.start_time = time.time()
        self.completed_ids.clear()
        self.objects = [self.classifier.classify(obj) for obj in objects]
    
    def get_next_target(self) -> Optional[ObjectRisk]:
        """根据当前阶段选择下一个目标"""
        if self.start_time is None:
            return None
        
        elapsed = time.time() - self.start_time
        remaining = self.time_limit - elapsed
        
        # 过滤已完成的物块
        candidates = [obj for obj in self.objects if obj.object_id not in self.completed_ids]
        
        if not candidates:
            return None
        
        # 根据阶段选择策略
        if elapsed < 7.0:
            # 阶段1：快速收割期
            return self._phase1_select(candidates)
        elif elapsed < 12.0:
            # 阶段2：稳定填充期
            return self._phase2_select(candidates, remaining)
        else:
            # 阶段3：保守收尾期
            return self._phase3_select(candidates, remaining)
    
    def _phase1_select(self, candidates: List[ObjectRisk]) -> Optional[ObjectRisk]:
        """阶段1：高置信度 + 近距离"""
        # 过滤：Safe + Medium
        filtered = [obj for obj in candidates 
                   if obj.risk_level in [RiskLevel.SAFE, RiskLevel.MEDIUM]]
        
        if not filtered:
            filtered = candidates  # 降级：如果没有Safe/Medium，考虑全部
        
        # 选择最近的
        return min(filtered, key=lambda obj: obj.estimated_time_s)
    
    def _phase2_select(self, candidates: List[ObjectRisk], remaining: float) -> Optional[ObjectRisk]:
        """阶段2：TSP路径优化（贪心最近邻）"""
        # 只考虑Safe物块
        safe_objs = [obj for obj in candidates if obj.risk_level == RiskLevel.SAFE]
        
        if not safe_objs:
            # 降级：如果没有Safe，考虑Medium
            safe_objs = [obj for obj in candidates if obj.risk_level == RiskLevel.MEDIUM]
        
        if not safe_objs:
            return None
        
        # TSP贪心：选择距离当前位置最近的Safe物块
        # TODO: 改进为真正的TSP近似算法
        return min(safe_objs, key=lambda obj: obj.estimated_time_s)
    
    def _phase3_select(self, candidates: List[ObjectRisk], remaining: float) -> Optional[ObjectRisk]:
        """阶段3：保守收尾（只选超高置信度且快速的）"""
        # 严格过滤：confidence>0.9 且 time<1.5s
        ultra_safe = [obj for obj in candidates 
                     if obj.confidence > 0.9 and obj.estimated_time_s < 1.5]
        
        if not ultra_safe:
            return None  # 宁可不抓，也不冒险触发15s判罚
        
        # 选择最快的
        return min(ultra_safe, key=lambda obj: obj.estimated_time_s)
    
    def mark_completed(self, object_id: int, success: bool):
        """标记物块完成状态，更新成功率"""
        self.completed_ids.add(object_id)
        
        # 找到对应物块并更新成功率
        for obj in self.objects:
            if obj.object_id == object_id:
                self.classifier.update_success_rate(obj.class_id, success)
                if not success:
                    obj.failure_count += 1
                break
    
    def should_abort(self) -> bool:
        """判断是否应该提前终止（避免15s判罚）"""
        if self.start_time is None:
            return False
        
        elapsed = time.time() - self.start_time
        remaining = self.time_limit - elapsed
        
        # 剩余时间<3s，终止
        return remaining < 3.0
```

### Step 3: 集成到主控制流程（2h）

**文件位置**：`电控/src/electrical_control/integration/control_flow.py`

```python
from electrical_control.scheduler.phased_scheduler import PhasedScheduler

class ControlFlow:
    def __init__(self):
        # ... 原有初始化
        self.scheduler = PhasedScheduler(time_limit_s=15.0)
    
    def on_batch_received(self, batch_data: dict):
        """收到视觉批量检测结果"""
        objects = batch_data['detections']
        
        # 启动调度器
        self.scheduler.start_round(objects)
        
        # 选择第一个目标
        self._dispatch_next_target()
    
    def _dispatch_next_target(self):
        """调度下一个目标"""
        # 检查是否应该提前终止
        if self.scheduler.should_abort():
            print("[SCHED] 剩余时间不足，提前终止避免15s判罚")
            self._finish_round()
            return
        
        # 选择下一个目标
        target = self.scheduler.get_next_target()
        
        if target is None:
            print("[SCHED] 无可用目标，本轮结束")
            self._finish_round()
            return
        
        # 发送任务到STM32
        task = self._build_item_task(target)
        self._send_to_stm32(task)
    
    def on_item_result(self, result: dict):
        """收到单件结果"""
        object_id = result['object_id']
        success = result['success']
        
        # 更新调度器
        self.scheduler.mark_completed(object_id, success)
        
        # 调度下一个
        self._dispatch_next_target()
```

### Step 4: TSP近似算法优化（可选，3h）

**文件位置**：`电控/src/electrical_control/scheduler/tsp_solver.py`

```python
from typing import List, Tuple

def greedy_tsp(objects: List[ObjectRisk], start_pos: Tuple[int, int]) -> List[ObjectRisk]:
    """贪心最近邻TSP近似算法"""
    if not objects:
        return []
    
    result = []
    remaining = objects.copy()
    current_pos = start_pos
    
    while remaining:
        # 找到距离当前位置最近的物块
        nearest = min(remaining, key=lambda obj: 
                     ((obj.position[0] - current_pos[0])**2 + 
                      (obj.position[1] - current_pos[1])**2)**0.5)
        
        result.append(nearest)
        remaining.remove(nearest)
        current_pos = nearest.position
    
    return result
```

    # 返回路径中的第一个

## 🚨 风险与缓解

**风险1**：时间估算不准确
- 缓解：实测每类物块的平均时间，动态校准

**风险2**：TSP算法性能开销大
- 缓解：限制候选物块数量（<10个），或用简化贪心

**风险3**：连续失败导致死循环
- 缓解：失败3次后降低优先级，但不跳过

**风险4**：阶段切换时机不当
- 缓解：可配置阶段时间阈值（7s/12s），现场调整
