#### Step 2: 电脑端监控GUI开发（10h）

**文件位置**：`工具/monitor_gui.py`

```python
#!/usr/bin/env python3
import sys
import socket
import json
from PyQt5.QtWidgets import *
from PyQt5.QtCore import *
from PyQt5.QtGui import *

class MaixCAMMonitor(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("MaixCAM实时监控")
        self.setGeometry(100, 100, 1200, 800)
        
        # 连接状态
        self.connected = False
        self.sock = None
        
        self.init_ui()
        self.connect_maixcam()
        
    def init_ui(self):
        central = QWidget()
        self.setCentralWidget(central)
        layout = QVBoxLayout(central)
        
        # 第1行：连接状态
        status_layout = QHBoxLayout()
        self.status_label = QLabel("未连接")
        self.status_label.setStyleSheet("color: red; font-weight: bold;")
        status_layout.addWidget(QLabel("状态:"))
        status_layout.addWidget(self.status_label)
        status_layout.addStretch()
        layout.addLayout(status_layout)
        
        # 第2行：参数调整
        param_group = QGroupBox("参数调整")
        param_layout = QFormLayout()
        
        self.conf_slider = QSlider(Qt.Horizontal)
        self.conf_slider.setRange(10, 95)
        self.conf_slider.setValue(50)
        self.conf_label = QLabel("0.50")
        self.conf_slider.valueChanged.connect(self.on_conf_changed)
        
        self.iou_slider = QSlider(Qt.Horizontal)
        self.iou_slider.setRange(10, 95)
        self.iou_slider.setValue(45)
        self.iou_label = QLabel("0.45")
        self.iou_slider.valueChanged.connect(self.on_iou_changed)
        
        conf_row = QHBoxLayout()
        conf_row.addWidget(self.conf_slider)
        conf_row.addWidget(self.conf_label)
        
        iou_row = QHBoxLayout()
        iou_row.addWidget(self.iou_slider)
        iou_row.addWidget(self.iou_label)
        
        param_layout.addRow("置信度阈值:", conf_row)
        param_layout.addRow("IOU阈值:", iou_row)
        
        self.apply_btn = QPushButton("应用参数")
        self.apply_btn.clicked.connect(self.apply_params)
        param_layout.addRow("", self.apply_btn)
        
        param_group.setLayout(param_layout)
        layout.addWidget(param_group)
        
        # 第3行：实时数据
        data_group = QGroupBox("实时数据")
        data_layout = QFormLayout()
        
        self.fps_label = QLabel("0.0")
        self.det_count_label = QLabel("0")
        self.last_update_label = QLabel("--")
        
        data_layout.addRow("FPS:", self.fps_label)
        data_layout.addRow("检测数量:", self.det_count_label)
        data_layout.addRow("最后更新:", self.last_update_label)
        
        data_group.setLayout(data_layout)
        layout.addWidget(data_group)
        
        # 第4行：检测结果表格
        self.det_table = QTableWidget()
        self.det_table.setColumnCount(4)
        self.det_table.setHorizontalHeaderLabels(["类别", "置信度", "坐标X", "坐标Y"])
        layout.addWidget(QLabel("检测结果:"))
        layout.addWidget(self.det_table)
        
    def connect_maixcam(self, host='172.20.10.2', port=9001):
        """连接MaixCAM监控端口"""
        try:
            self.sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            self.sock.connect((host, port))
            self.connected = True
            self.status_label.setText("已连接")
            self.status_label.setStyleSheet("color: green; font-weight: bold;")
            
            # 启动接收线程
            self.recv_thread = RecvThread(self.sock)
            self.recv_thread.data_received.connect(self.on_data_received)
            self.recv_thread.start()
            
        except Exception as e:
            QMessageBox.critical(self, "连接失败", f"无法连接到MaixCAM: {e}")
    
    def on_conf_changed(self, value):
        self.conf_label.setText(f"{value/100:.2f}")
    
    def on_iou_changed(self, value):
        self.iou_label.setText(f"{value/100:.2f}")
    
    def apply_params(self):
        """发送参数更新到MaixCAM"""
        if not self.connected:
            QMessageBox.warning(self, "未连接", "请先连接MaixCAM")
            return
        
        cmd = {
            'type': 'UPDATE_PARAMS',
            'conf_th': self.conf_slider.value() / 100,
            'iou_th': self.iou_slider.value() / 100
        }
        
        try:
            self.sock.sendall(json.dumps(cmd).encode())
            QMessageBox.information(self, "成功", "参数已发送")
        except Exception as e:
            QMessageBox.critical(self, "发送失败", f"参数发送失败: {e}")
    
    def on_data_received(self, data):
        """处理接收到的监控数据"""
        self.fps_label.setText(f"{data['fps']:.1f}")
        self.det_count_label.setText(str(len(data['detections'])))
        self.last_update_label.setText(QTime.currentTime().toString())
        
        # 更新检测结果表格
        self.det_table.setRowCount(len(data['detections']))
        for i, det in enumerate(data['detections']):
            self.det_table.setItem(i, 0, QTableWidgetItem(str(det['class_id'])))
            self.det_table.setItem(i, 1, QTableWidgetItem(f"{det['confidence']:.2f}"))
            self.det_table.setItem(i, 2, QTableWidgetItem(str(det['bbox'][0])))
            self.det_table.setItem(i, 3, QTableWidgetItem(str(det['bbox'][1])))

class RecvThread(QThread):
    data_received = pyqtSignal(dict)
    
    def __init__(self, sock):
        super().__init__()
        self.sock = sock
        self.running = True
    
    def run(self):
        buffer = b''
        while self.running:
            try:
                chunk = self.sock.recv(4096)
                if not chunk:
                    break
                buffer += chunk
                
                # 简单的换行分隔JSON
                while b'\n' in buffer:
                    line, buffer = buffer.split(b'\n', 1)
                    data = json.loads(line)
                    self.data_received.emit(data)
            except:
                break

if __name__ == '__main__':
    app = QApplication(sys.argv)
    window = MaixCAMMonitor()
    window.show()
    sys.exit(app.exec_())
```

### 风险与缓解

**风险1**：监控程序崩溃影响主流程
- 缓解：监控通道独立，崩溃不影响MaixCAM推理

**风险2**：网络延迟导致参数更新不及时
- 缓解：参数更新采用TCP，确认收到ACK再提示成功

**风险3**：比赛时忘记关闭监控
- 缓解：MaixCAM支持无监控运行，监控程序可选

## ❌ 不推荐方案：迁移模型到电脑

### 为什么不推荐

1. **违反已锁定架构**：方案B已确认MaixCAM本地推理
2. **赛规灰区风险**：外部计算可能被判定为遥控
3. **稳定性倒退**：新增笔记本重启、USB断连等故障点
4. **时间不足**：6天内无法完成完整验证
5. **收益低**：当前MaixCAM性能已足够
