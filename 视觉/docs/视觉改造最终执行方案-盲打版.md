```python
# 发送

send_task({
    'image_id': combined,  # 0-255
    'confidence': confidence
})
```

#### 6. 3点ToF测量（可选升级）（1天）

**触发条件**：若单点ToF消歧准确率<83%

**方案**：
```python
def read_3point_depth(tof_sensor, bbox):
    """测量中心+左右偏移10mm"""
    center = bbox.center()
    left = (center[0] - 10, center[1])
    right = (center[0] + 10, center[1])
    
    h_center = tof_sensor.read_at(center)
    h_left = tof_sensor.read_at(left)
    h_right = tof_sensor.read_at(right)
    
    # 平顶判据
    if abs(h_center - h_left) < 5 and abs(h_center - h_right) < 5:
        return "flat_top"  # 圆柱/正方体
    else:
        return "pointed_top"  # 圆锥/四棱锥
```

**注意**：需要可移动支架或3个ToF传感器

```python
def hybrid_shape_classification(yolo_result, tof_reading, shape_features):
    """混合分类器"""
    if tof_reading.confidence > 0.9:
        return depth_disambiguate(...)  # 高置信度，信任ToF
    
    elif tof_reading.confidence > 0.5:
        # 中等置信度，加权融合
        tof_conf = tof_reading.confidence
        feature_conf = compute_feature_confidence(shape_features)
        
        final_conf = 0.7 * tof_conf + 0.3 * feature_conf
        if final_conf > 0.7:
            return depth_disambiguate(...)
        else:
            return yolo_result.class_name  # 降级到YOLO
    
    else:
        # 低置信度，用形状特征
        return classify_shape_contour(shape_features)
```
